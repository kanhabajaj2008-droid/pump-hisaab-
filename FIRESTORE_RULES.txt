rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function me() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function owner() { return signedIn() && me().role == 'owner' && me().active == true; }
    function staffOrOwner() { return signedIn() && me().active == true; }
    function canAccessPump(pumpId) {
      return owner() || (pumpId in me().pumpAccess);
    }

    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || owner());
      allow create: if owner();
      allow update: if owner();
      allow delete: if owner();
    }

    match /pumps/{pumpId} {
      allow read: if signedIn() && canAccessPump(pumpId);
      allow write: if owner();
    }

    match /fuelPrices/{pumpId} {
      allow read: if signedIn() && canAccessPump(pumpId);
      allow create, update, delete: if owner() && canAccessPump(pumpId);
    }

    match /dailyShifts/{shiftId} {
      allow read: if signedIn() && canAccessPump(resource.data.pumpId);
      allow create: if staffOrOwner() && canAccessPump(request.resource.data.pumpId) && request.resource.data.createdBy == request.auth.uid;
      allow update: if signedIn() && canAccessPump(resource.data.pumpId) && (owner() || resource.data.createdBy == request.auth.uid);
      allow delete: if owner();
    }

    match /{document=**} { allow read, write: if false; }
  }
}
